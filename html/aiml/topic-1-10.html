The .env File – Managing Secrets Professionally

bash 
# NEVER hardcode secrets in Python files!
# BAD (dangerous, do not do this):
API_KEY = "sk-live-abc123xyz"
DB_PASSWORD = "mySecretPassword"

# GOOD: Use .env file + python-dotenv:
# .env file (add to .gitignore!):
API_KEY=sk-live-abc123xyz
DB_HOST=localhost
DB_NAME=myapp
DB_PASSWORD=mySecretPassword
DEBUG=True

# Install python-dotenv:
pip install python-dotenv
python 
# main.py - load from .env:
from dotenv import load_dotenv
import os

load_dotenv()  # reads .env file and loads into environment

api_key  = os.getenv("API_KEY")
db_host  = os.getenv("DB_HOST", "localhost")  # default value if not set
debug    = os.getenv("DEBUG", "False").lower() == "true"

print(f"API Key: {api_key[:8]}...")    # only show first 8 chars!
print(f"DB Host: {db_host}")
print(f"Debug mode: {debug}")
