Complete Cheat Sheet – Python Environment Setup

bash 
# INSTALLATION CHECK:
python --version                     # should be 3.11+
python3 --version                    # (Mac/Linux)
pip --version                        # verify pip works

# CREATE PROJECT ENVIRONMENT:
mkdir my_project && cd my_project
python -m venv venv                  # create environment
python -m pip install --upgrade pip  # update pip

# ACTIVATE:
source venv/bin/activate             # macOS/Linux
venv\Scripts\activate                # Windows CMD
venv\Scripts\Activate.ps1            # Windows PowerShell
# Prompt shows: (venv) $

# VERIFY ACTIVATION:
which python                         # Mac/Linux: should show venv/bin/python
where python                         # Windows: should show venv\Scripts\python.exe
python -c "import sys; print(sys.prefix)"  # shows venv path

# INSTALL PACKAGES:
pip install requests                 # latest
pip install requests==2.31.0        # exact version
pip install requests pandas numpy   # multiple at once
pip install -r requirements.txt     # from file

# VIEW PACKAGES:
pip list                             # all packages
pip show requests                    # details about one package
pip list --outdated                  # packages with newer versions

# FREEZE / REQUIREMENTS:
pip freeze > requirements.txt        # generate (exact versions)
cat requirements.txt                 # view it
pip install -r requirements.txt      # recreate environment

# UPGRADE / REMOVE:
pip install --upgrade requests       # upgrade
pip uninstall requests               # remove

# DEACTIVATE:
deactivate                           # leave virtual environment

# .gitignore MUST INCLUDE:
# venv/
# __pycache__/
# *.pyc
# .env
python 
# VERIFY PYTHON VERSION IN SCRIPT:
import sys
print(sys.version)                  # 3.11.5 (main, ...)
print(sys.version_info.major)       # 3
in_venv = hasattr(sys, "real_prefix") or sys.base_prefix != sys.prefix
print(in_venv)                      # True (if venv is active)
